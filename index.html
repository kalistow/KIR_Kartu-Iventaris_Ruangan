Gagal mengambil data. Pastikan Anda telah membuat tabel `master_bmd` dan relasi di tabel `assets` pada Editor SQL Supabase Anda. Gunakan perintah SQL berikut:

-- Salakan copy dan jalankan di Supabase SQL Editor:
create table public.master_bmd (
    id bigint generated always as identity primary key,
    nibar text unique,
    no_register text,
    kode_barang text,
    nama_barang text,
    spesifikasi_nama text,
    spesifikasi_lainnya text,
    merek_tipe text,
    nomor_polisi text,
    nomor_rangka text,
    nomor_bpkb text,
    jumlah integer,
    satuan text,
    harga numeric,
    nilai numeric,
    cara_perolehan text,
    tanggal_perolehan date,
    status_penggunaan text,
    keterangan text,
    status text default 'Aktif',
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

create table public.assets (
    id bigint generated always as identity primary key,
    master_bmd_id bigint references public.master_bmd(id) on delete set null,
    no_urut integer,
    kode_barang text,
    nama_barang text,
    merk_model text,
    no_seri text,
    ukuran text,
    bahan text,
    tahun text,
    jumlah integer,
    harga numeric,
    ruangan text default 'Belum Terpetakan',
    kondisi text default 'Baik',
    keterangan text,
    status text default 'Aktif',
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

create table public.riwayat_barang (
    id bigint generated always as identity primary key,
    asset_id bigint references public.assets(id) on delete cascade,
    jenis_perubahan text,
    keterangan text,
    tanggal timestamp with time zone default timezone('utc'::text, now()) not null
);

alter table public.master_bmd disable row level security;
alter table public.assets disable row level security;
alter table public.riwayat_barang disable row level security;

-- Tabel untuk log error import (Sesuai instruksi)
create table public.import_logs (
    id bigint generated always as identity primary key,
    tanggal timestamp with time zone default timezone('utc'::text, now()) not null,
    jenis_file text,
    pesan_error text,
    detail jsonb
);
alter table public.import_logs disable row level security;

-- Pastikan User Supabase API memiliki izin INSERT, UPDATE, DELETE (Sesuai instruksi)
GRANT INSERT, UPDATE, DELETE ON public.master_bmd TO anon, authenticated;
GRANT INSERT, UPDATE, DELETE ON public.import_logs TO anon, authenticated;
